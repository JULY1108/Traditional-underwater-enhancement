cmake_minimum_required(VERSION 3.14)
project(underwater_enhancement LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
add_library(underwater_enhancement dehaze_old.cpp smooth_clahe.cpp)
target_include_directories(underwater_enhancement PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(underwater_enhancement PUBLIC ${OpenCV_LIBS} OpenMP::OpenMP_CXX Threads::Threads)
if(MSVC)
    target_compile_options(underwater_enhancement PUBLIC /utf-8)
endif()
add_executable(demo_images demo_TSA.cpp)
target_link_libraries(demo_images PRIVATE underwater_enhancement)
